        let currentIdx = 0;
        const userAnswers = new Array(questions.length).fill("");

        const questionCounter = document.getElementById("question-counter");
        const questionTag = document.getElementById("question-tag");
        const questionText = document.getElementById("question-text");
        const userAnswerInput = document.getElementById("user-answer");
        const feedbackBox = document.getElementById("feedback-box");
        const progressBar = document.getElementById("progress-bar");
        const prevBtn = document.getElementById("prev-btn");
        const submitBtn = document.getElementById("submit-btn");
        const nextBtn = document.getElementById("next-btn");
        const quizContainer = document.getElementById("quiz-container");
        const resultsScreen = document.getElementById("results-screen");
        const scoreSummary = document.getElementById("score-summary");

        function loadQuestion() {
            const q = questions[currentIdx];
            questionCounter.innerText = `السؤال ${currentIdx + 1} من ${questions.length}`;
            questionTag.innerText = q.tag;
            questionText.innerText = q.question;
            userAnswerInput.value = userAnswers[currentIdx] || "";
            
            // Update progress
            progressBar.style.width = `${((currentIdx + 1) / questions.length) * 100}%`;
            
            // Buttons state
            prevBtn.disabled = currentIdx === 0;
            
            // Check if already answered
            if (userAnswers[currentIdx]) {
                showFeedback(q.modelAnswer);
            } else {
                feedbackBox.classList.add("hidden");
                submitBtn.classList.remove("hidden");
                nextBtn.classList.add("hidden");
                userAnswerInput.disabled = false;
            }
        }

        function checkAnswer() {
            const val = userAnswerInput.value.trim();
            if (!val) {
                alert("الرجاء كتابة إجابتك قبل التحقق.");
                return;
            }
            userAnswers[currentIdx] = val;
            showFeedback(questions[currentIdx].modelAnswer);
        }

        function showFeedback(model) {
            feedbackBox.innerHTML = `<strong class="block text-emerald-400 mb-1 font-semibold">الإجابة النموذجية والمعيارية:</strong><p class="text-slate-300">${model}</p>`;
            feedbackBox.className = "p-4 rounded-xl border bg-slate-900/80 border-emerald-500/30 text-sm leading-relaxed block";
            submitBtn.classList.add("hidden");
            nextBtn.classList.remove("hidden");
            userAnswerInput.disabled = true;
        }

        function nextQuestion() {
            if (currentIdx < questions.length - 1) {
                currentIdx++;
                loadQuestion();
            } else {
                showResults();
            }
        }

        function prevQuestion() {
            if (currentIdx > 0) {
                currentIdx--;
                loadQuestion();
            }
        }

        function showResults() {
            quizContainer.classList.add("hidden");
            resultsScreen.classList.remove("hidden");
            scoreSummary.innerText = `لقد اجتزت الأسئلة العشرة الصعبة والمتقدمة في قواعد النحو والبلاغة بنجاح. راجع إجاباتك النموذجية في كل سؤال لتثبيت المعرفة الدقيقة.`;
        }

        function restartQuiz() {
            currentIdx = 0;
            userAnswers.fill("");
            resultsScreen.classList.add("hidden");
            quizContainer.classList.remove("hidden");
            loadQuestion();
        }
